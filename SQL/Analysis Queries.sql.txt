-- =========================================================
-- DATA PREVIEW
-- =========================================================

SELECT *
FROM `portfolio-projects-495215.Ecommerce_data.order_data`
LIMIT 10;

-- Business Insight:
-- Displays the first 10 records from the order_data table
-- to verify that the dataset has been imported correctly.

-- =========================================================
-- KPI ANALYSIS
-- =========================================================

-- Total Revenue

SELECT
    ROUND(SUM(payment_value), 2) AS Total_revenue
FROM `portfolio-projects-495215.Ecommerce_data.order_payment`;

-- Business Insight:
-- Calculates the total revenue generated from all customer payments.


-- Total Orders

SELECT
    COUNT(DISTINCT order_id) AS Total_Order
FROM `portfolio-projects-495215.Ecommerce_data.order_payment`;

-- Business Insight:
-- Calculates the total number of unique customer orders.


-- Average Order Value

SELECT
    ROUND(AVG(payment_value), 2) AS Avg_ordervalue
FROM `portfolio-projects-495215.Ecommerce_data.order_payment`;

-- Business Insight:
-- Calculates the average amount spent per order.

-- =========================================================
-- ORDER ANALYSIS
-- =========================================================

-- Order Status Analysis

SELECT
    order_status,
    COUNT(*) AS Total_orders
FROM `portfolio-projects-495215.Ecommerce_data.order_data`
GROUP BY order_status
ORDER BY Total_orders DESC;

-- Business Insight:
-- Displays the number of orders for each order status.


-- Monthly Sales Trend

SELECT
    FORMAT_DATE('%y-%m', order_purchase_timestamp) AS Monthly_Order,
    ROUND(SUM(payment_value), 2) AS Revenue
FROM `portfolio-projects-495215.Ecommerce_data.order_data` AS o
JOIN `portfolio-projects-495215.Ecommerce_data.order_payment` AS p
ON o.order_id = p.order_id
GROUP BY Monthly_Order
ORDER BY Monthly_Order DESC;

-- Business Insight:
-- Analyses monthly revenue trends by combining order and payment data.

-- =========================================================
-- PRODUCT ANALYSIS
-- =========================================================

-- Top 10 Product Categories by Revenue

SELECT
    product_category_name,
    ROUND(SUM(price), 2) AS Revenue
FROM `portfolio-projects-495215.Ecommerce_data.product_data` AS p
JOIN `portfolio-projects-495215.Ecommerce_data.order_item` AS oi
ON p.product_id = oi.product_id
GROUP BY product_category_name
ORDER BY Revenue DESC
LIMIT 10;

-- Business Insight:
-- Identifies the top 10 product categories generating the highest sales revenue.


-- Products Above Average Price (Subquery)

SELECT
    product_id,
    price
FROM `portfolio-projects-495215.Ecommerce_data.order_item`
WHERE price >
(
    SELECT AVG(price)
    FROM `portfolio-projects-495215.Ecommerce_data.order_item`
)
ORDER BY price DESC;

-- Business Insight:
-- Returns products priced above the overall average product price.

-- =========================================================
-- SELLER ANALYSIS
-- =========================================================

-- Top 10 Sellers by Revenue

SELECT
    seller_id,
    ROUND(SUM(price), 2) AS Revenue
FROM `portfolio-projects-495215.Ecommerce_data.order_item`
GROUP BY seller_id
ORDER BY Revenue DESC
LIMIT 10;

-- Business Insight:
-- Identifies the highest revenue-generating sellers.


-- Sellers with Revenue Above Average (Subquery)

SELECT
    seller_id,
    ROUND(SUM(price), 2) AS seller_revenue
FROM `portfolio-projects-495215.Ecommerce_data.order_item`
GROUP BY seller_id
HAVING seller_revenue >
(
    SELECT AVG(seller_total)
    FROM
    (
        SELECT
            SUM(price) AS seller_total
        FROM `portfolio-projects-495215.Ecommerce_data.order_item`
        GROUP BY seller_id
    )
)
ORDER BY seller_revenue DESC;

-- Business Insight:
-- Returns sellers whose total revenue is greater than the average revenue generated by all sellers.

-- =========================================================
-- CUSTOMER ANALYSIS
-- =========================================================

-- Customer Distribution by State

SELECT
    customer_state,
    COUNT(*) AS Total_customer
FROM `portfolio-projects-495215.Ecommerce_data.customer_data`
GROUP BY customer_state
ORDER BY Total_customer DESC;

-- Business Insight:
-- Shows the distribution of customers across different Brazilian states.


-- Customers Above Average Spending (Subquery)

SELECT
    customer_unique_id,
    ROUND(SUM(payment_value), 2) AS Total_spent
FROM `portfolio-projects-495215.Ecommerce_data.customer_data` AS c
JOIN `portfolio-projects-495215.Ecommerce_data.order_data` AS o
ON c.customer_id = o.customer_id
JOIN `portfolio-projects-495215.Ecommerce_data.order_payment` AS p
ON o.order_id = p.order_id
GROUP BY customer_unique_id
HAVING Total_spent >
(
    SELECT AVG(payment_value)
    FROM `portfolio-projects-495215.Ecommerce_data.order_payment`
)
ORDER BY Total_spent DESC;

-- Business Insight:
-- Identifies customers whose total spending is higher than the overall average payment value.

-- =========================================================
-- DELIVERY ANALYSIS
-- =========================================================

-- Average Delivery Time

SELECT
    ROUND(
        AVG(
            DATE_DIFF(
                order_delivered_customer_date,
                order_purchase_timestamp,
                DAY
            )
        )
    ) AS Avg_DeliveryDays
FROM `portfolio-projects-495215.Ecommerce_data.order_data`
WHERE order_delivered_customer_date IS NOT NULL;

-- Business Insight:
-- Calculates the average number of days required to deliver customer orders.
